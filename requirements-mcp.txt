# MCP-Server von postmakr fuer den eigenen Rechner (mcp_server/).
# Installieren, am besten in ein venv:
#   python3 -m venv .venv && . .venv/bin/activate
#   pip install -r requirements-mcp.txt
#   python3 -m playwright install chromium   # einmalig, fuer render_slide
# ffmpeg muss zusaetzlich auf dem System sein (Animationen, Ausschnitte aus dem
# Lager, Hoercheck), auf dem Mac etwa mit: brew install ffmpeg

# -- Grundausstattung ----------------------------------------------------
fastmcp==3.4.4          # MCP-Server (stdio und HTTP), bringt mcp, starlette, httpx mit
uvicorn==0.51.0         # nur fuer den HTTP-Modus (--http)
playwright==1.61.0      # Slide-Rendering ueber engine-shared/tools/render_static/render.py
feedparser==6.0.12      # scan/fetch_news.py (RSS)
PyYAML==6.0.3           # brand.yaml, meta.yaml, sounds.yaml und so weiter
numpy>=1.26             # Such-Index und Hoercheck

# -- Modelle (optional) ---------------------------------------------------
# Die Modelle fuer Suche nach Bedeutung und Bild, Transkripte und Hoercheck
# stehen in requirements-extras.txt und kommen nur mit
# ./postmakr installieren --extras dazu. Ohne sie laufen alle Tools trotzdem:
# Die Suche nutzt dann nur Stichworte, der Hoercheck misst Soll-Liste und
